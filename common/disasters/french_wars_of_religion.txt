french_wars_of_religion = {
	potential = {
		tag = FRA
		religion_group = christian
		NOT = { has_country_flag = FRA_wars_of_religion_ended }
		OR = {
			current_age = age_of_discovery
			current_age = age_of_reformation
		}
	}
	
	can_start = {
		tag = FRA
		current_age = age_of_reformation
		is_religion_enabled = reformed
		has_any_disaster = no
		any_owned_province = {
			superregion = europe_superregion
			religion_group = christian
			has_owner_religion = no
		}
	}
	
	can_stop = {
		has_any_disaster = yes
	}
	
	progress = {
		modifier = {
			factor = 1
			NOT = { religious_unity = 0.9 }
			religious_unity = 0.75
		}
		modifier = {
			factor = 2
			NOT = { religious_unity = 0.8 }
		}
		modifier = {
			factor = 5
			NOT = { religious_unity = 0.7 }
		}
		modifier = {
			factor = 5
			has_idea_group = religious_ideas
		}
		modifier = {
			factor = 1
			NOT = { stability = 2 }
		}
		modifier = {
			factor = 2
			NOT = { stability = 1 }
		}
		modifier = {
			factor = 5
			NOT = { stability = 0 }
		}
		modifier = {
			factor = 2
			has_hostile_reformation_center = yes
		}
	}
	
	can_end = {
		NOT = { num_of_rebel_controlled_provinces = 1 }
		NOT = { num_of_rebel_armies = 1 }
		stability = 1
		capital_scope = {
			controlled_by = ROOT
		}
		custom_trigger_tooltip = {
			tooltip = disaster_active_for_10_years_tooltip
			had_country_flag = {
				flag = FRA_wars_of_religion_started
				days = 3650
			}
		}
		NOT = {
			france_region = {
				owned_by = ROOT
				has_owner_religion = no
				unrest = 5
			}
		}
	}

	on_start = french_wars_of_religion.1
	on_end = french_wars_of_religion.100
	
	on_monthly = {
		random_events = { 
			1000 = 0
			200 = french_wars_of_religion.2
			50 = french_wars_of_religion.6
			50 = french_wars_of_religion.8
		}
	}
}
