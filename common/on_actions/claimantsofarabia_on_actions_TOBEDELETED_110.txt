on_new_monarch = {
	# Seljuks
	if = {
		limit = {
			has_country_flag = sel_mission_unlocked
			NOT = { mission_completed = seljuk_arslan_mission }
		}
		country_event = {
			id = ab_seljuk.31
		}
	}
	if = {
		limit = {
			mission_completed = seljuk_arslan_mission
		}
		add_ruler_personality = arslan_descendant_personality
	}
	# Arabia
	if = {
		limit = {
			has_country_flag = ab_bloodline_muhammad
		}
		add_ruler_personality = muhammad_descendant_personality
	}
	# Ayyubids
	if = {
		limit = {
			mission_completed = hsn_depose_nizzarids
		}
		add_ruler_personality = saladin_descendant_personality
	}
	if = {
		limit = {
			mission_completed = hsn_secure_succession
		}
		country_event = {
			id = ab_ayyubids.8
		}
	}
}

on_advisor_hired = {

		}
		remove_country_modifier = sel_hiring_advisor_mod
		add_country_modifier = {
			name = sel_hiring_advisor_mod
			duration = 1095
		}
	}
	# Nizzarids
	if = {
		limit = {
			has_country_flag = naz_hiring_advisor_bonus
		}
		remove_country_modifier = naz_hiring_advisor_mod
		add_country_modifier = {
			name = naz_hiring_advisor_mod
			duration = 1095
		}
	}
	# Assyria
	if = {
		limit = {
			has_faction = faction_asr_nestorian
		}
		add_faction_influence = {
			faction = faction_asr_nestorian
			influence = 3
		}
	}
	# Chorath
	if = {
		limit = {
			has_country_flag = xbh_hiring_advisor
			OR = {
				employed_advisor = {
					category = ADM
					religion = sunni
				}
				employed_advisor = {
					category = DIP
					religion = sunni
				}
				employed_advisor = {
					category = MIL
					religion = sunni
				}
			}
		}
		remove_country_modifier = naz_hiring_advisor_mod
		add_country_modifier = {
			name = naz_hiring_advisor_mod
			duration = 180
		}
	}
	# Ayyubids
	if = {
		limit = {
			has_country_flag = hsn_hiring_advisor
			OR = {
				employed_advisor = {
					category = ADM
					culture = kurdish
				}
				employed_advisor = {
					category = DIP
					culture = kurdish
				}
				employed_advisor = {
					category = MIL
					culture = kurdish
				}
			}
		}
		remove_country_modifier = hsn_hiring_advisor
		add_country_modifier = {
			name = hsn_hiring_advisor
			duration = 730
		}
	}
}

on_province_culture_changed = {
	# Seljuks
	if = {
		limit = {
			owner = {
				has_country_flag = sel_accepted_all_persian
			}
		}
		if = {
			limit = {
				culture_group = iranian
				NOT = { has_owner_accepted_culture = yes }
				NOT = { has_province_modifier = jlr_persian_trappings }
			}
			add_province_modifier = {
				name = jlr_persian_trappings
				duration = -1
			}
		}
		if = {
			limit = {
				has_province_modifier = jlr_persian_trappings
				OR = {
					NOT = { culture_group = iranian }
					has_owner_accepted_culture = yes
				}
			}
			remove_province_modifier = jlr_persian_trappings
		}
	}
	# Assyria
	if = {
		limit = {
			has_province_modifier = nnv_spread_of_assyrian
		}
		remove_province_modifier = nnv_spread_of_assyrian
	}
	if = {
		limit = {
			owner = {
				has_country_flag = nnv_assyrian_persia
			}
			culture_group = iranian
		}
		add_province_modifier = {
			name = nnv_spread_of_assyrian
			duration = -1
		}
	}
	if = {
		limit = {
			owner = {
				has_country_flag = nnv_assyrian_anatolia
			}
			region = anatolia_region
			NOT = { culture = assyrian }
		}
		add_province_modifier = {
			name = nnv_spread_of_assyrian
			duration = -1
		}
	}
}
