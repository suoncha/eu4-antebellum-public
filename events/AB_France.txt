#Republic v. Monarchy
country_event = {
	id = ab_france.0
	title = ab_france.0.t
	desc = ab_france.0.d
	picture = GOOD_WITH_MONARCH_eventPicture
	
	is_triggered_only = yes

	option = { #Monarchy
		name = ab_france.0.a
		add_stability = -1
		add_estate_loyalty = {
			estate = estate_nobles
			loyalty = -25
		}
		random_province = {
			limit = {
				region = france_region
				owned_by = ROOT
			}
			spawn_rebels = {
				type = pretender_rebels
				size = 3
			}
		}
		set_country_flag = fra_monarchy_focus
		swap_non_generic_missions = yes
		country_event = { id = ab_france.1 }
		set_government_rank = 2
		add_government_reform = fraelective
		hidden_effect = {
			swap_free_idea_group = yes
		}
	}

	option = { #No Gods, No Kings, long live the republic
		name = ab_france.0.b
		set_country_flag = fra_communist_focus
		swap_non_generic_missions = yes
		define_ruler = {
			claim = 75
		}
		set_government_rank = 2
		change_government = republic
		add_government_reform = frarepublic
		hidden_effect = {
			set_global_flag = AB_FRA_REPUBLIC_RISE
			set_variable = {
				which = fraBattleCount
				value = 0
			}
			swap_free_idea_group = yes
			every_subject_country = {
				restore_country_color = yes
			}
			release_all_subjects = yes
		}
	}
}

				}
				change_primary_culture = french_culture
				set_heir_culture = french_culture
				set_ruler_culture = french_culture
			}
		}
	}
}

# This event is very special, every option of this event is converted into
# a vote that can be clicked on the Holy Roman Empire interface
country_event = {
	id = ab_france.101
	title = ab_france.101.t
	desc = ab_france.101.d
	picture = HRE_eventPicture

	fire_only_once = yes
	is_triggered_only = yes
	major = yes

	option = { #invade France
		name = ab_france.101.a
		country_event = {
			id = ab_france.102
		}
		custom_tooltip = AB_EMPTY
		FRA = {
			add_opinion = {
				who = emperor
				modifier = opinion_imperial_rejection
			}
		}
	}

	option = { #do nothing
		name = ab_france.101.b
		FRA = {
			add_prestige = 25
			add_republican_tradition = 25
			add_country_modifier = {
				name = ab_fra_imperial_recognition
				duration = 3650
			}
			add_opinion = {
				who = emperor
				modifier = opinion_imperial_recognition
			}
		}
		emperor = {
			add_country_modifier = {
				name = ab_fra_imperial_concede
				duration = 7300
			}
		}
	}
}

# Emperor deciding between Balkanizing France or Not
country_event = {

	desc = ab_france.102.d
	picture = HRE_eventPicture

	major = yes
	is_triggered_only = yes

	option = { #Force the Monarchy
		name = ab_france.102.a

		custom_tooltip = AB_FRA_REPUBLICAN_INCIDENT_RESTORE
		set_global_flag = ab_fra_republican_incident_monarchy_choice

			country_event = {
				id = ab_france.103
			}
		}

	option = { #Balkanize France instead
		name = ab_france.102.b
		ai_chance = { factor = 30 }
		custom_tooltip = AB_FRA_REPUBLICAN_INCIDENT_BALKANIZE

		declare_war_with_cb = {
			casus_belli = cb_fate_of_france
			who = FRA
