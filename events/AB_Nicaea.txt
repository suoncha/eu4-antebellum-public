	trigger = {
		OR = {
			tag = NCI
			was_tag = NCI
		}
		has_country_flag = nci_can_become_hellenic
		religion = orthodox
		OR = {
			employed_advisor = {
				category = ADM
				is_male = yes
				religion = hellenic
				culture = greek

			}
			has_country_flag = ab_nicaea_plethon

						every_owned_province = {
							limit = {
								religion = hellenic

								50 = {
									spawn_rebels = {
										size = 1

								}

						}
